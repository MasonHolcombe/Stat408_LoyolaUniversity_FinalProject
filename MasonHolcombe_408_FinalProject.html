Data Processing

cols_to_remove = c("gameid", "datacompleteness", "url", "league", "year",
                   "split", "date", "game", "participantid",
                   "playername", "playerid", "teamname", "teamid",
                   "ban1", "ban2", "ban3", "ban4", "ban5",
                   "pick1", "pick2", "pick3", "pick4", "pick5",
                   "result")
X = lol25 %>% 
  select(-all_of(cols_to_remove)) %>%
  mutate(across(everything(), ~ replace_na(., 0)))
X = model.matrix(~ . - 1, data = X)
X_df = as.data.frame(X)

y = as.numeric(lol25$result)

LASSO CV

lasso_cv = cv.glmnet(
  X, y,
  alpha = 1,
  family = "binomial",
  nlambda = 30,
  nfolds = 5,
)
plot(lasso_cv)

Optimal Lambda

best_lambda = lasso_cv$lambda.min
model_best = glmnet(X, y, alpha = 1, lambda = best_lambda, family = "binomial")

All Coefficients

coef_df = as.data.frame(as.matrix(coef(model_best)))
coef_df$feature = rownames(coef_df)
colnames(coef_df)[1] = "coefficient"

Kills, Deaths, Assists by Result & Position

df_long = lol25 %>%
  filter(position != "team") %>%
  select(result, position, kills, deaths, assists) %>%
  pivot_longer(
    cols = c(kills, deaths, assists),
    names_to = "stat",
    values_to = "value"
  ) %>% 
  mutate(
    stat = factor(stat,
                  levels = c("kills", "deaths", "assists"))
  )

ggplot(df_long, aes(x = position, y = value, fill = stat)) +
  geom_boxplot() +
  facet_wrap(~ result,
             labeller = as_labeller(c(`0` = "Loss",
                                      `1` = "Win"))) +
  labs(
    title = "Kills, Deaths, and Assists by Result & Position",
    x = "Position",
    y = "Value"
  ) +
  scale_fill_manual(values = c(
    "kills" = "skyblue",
    "deaths" = "salmon",
    "assists" = "lightgreen"
